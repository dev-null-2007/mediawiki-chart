1. Get your MediaWiki login credentials by running:

    echo Username: {{ .Values.mediawikiUser }}
    echo Password: $(microk8s kubectl get secret --namespace {{ .Release.Namespace }} {{ template "common.names.fullname" . }} -o jsonpath="{.data.mediawiki-password}" | base64 -d)

2. Access MediaWiki:

{{- if .Values.ingress.enabled }}

   Visit: http{{ if .Values.ingress.tls }}s{{ end }}://{{ .Values.ingress.hostname }}

{{- else if contains "NodePort" .Values.service.type }}

   export NODE_PORT=$(microk8s kubectl get --namespace {{ .Release.Namespace }} -o jsonpath="{.spec.ports[0].nodePort}" services {{ template "common.names.fullname" . }})
   export NODE_IP=$(microk8s kubectl get nodes --namespace {{ .Release.Namespace }} -o jsonpath="{.items[0].status.addresses[0].address}")
   echo "MediaWiki URL: http://$NODE_IP:$NODE_PORT/"

{{- else if contains "LoadBalancer" .Values.service.type }}

   NOTE: It may take a few minutes for the LoadBalancer IP to be available.
         Watch the status with: 'microk8s kubectl get svc --namespace {{ .Release.Namespace }} -w {{ template "common.names.fullname" . }}'

   export SERVICE_IP=$(microk8s kubectl get svc --namespace {{ .Release.Namespace }} {{ template "common.names.fullname" . }} --template "{{ "{{ range (index .status.loadBalancer.ingress 0) }}{{ . }}{{ end }}" }}")
   {{- $port:= ( coalesce .Values.service.ports.http .Values.service.port) | toString }}
   echo "MediaWiki URL: http://$SERVICE_IP{{- if ne $port "80" }}:{{ coalesce .Values.service.ports.http .Values.service.port }}{{ end }}/"

{{- else if contains "ClusterIP" .Values.service.type }}

   Use port-forward to access MediaWiki:

   microk8s kubectl port-forward --namespace {{ .Release.Namespace }} svc/{{ template "common.names.fullname" . }} 8080:{{ coalesce .Values.service.ports.http .Values.service.port }}

   Then visit: http://localhost:8080

{{- end }}

4. Upgrading this release:

   Passwords are automatically managed. To upgrade:

   microk8s helm upgrade {{ .Release.Name }} ./mediawiki/

   If you get password validation errors during upgrade, you can explicitly provide them:

   MARIADB_ROOT_PASSWORD=$(microk8s kubectl get secret --namespace {{ .Release.Namespace }} {{ template "mediawiki.mariadb.fullname" . }} -o jsonpath="{.data.mariadb-root-password}" | base64 -d)
   MARIADB_PASSWORD=$(microk8s kubectl get secret --namespace {{ .Release.Namespace }} {{ template "mediawiki.mariadb.fullname" . }} -o jsonpath="{.data.mariadb-password}" | base64 -d)
   MEDIAWIKI_PASSWORD=$(microk8s kubectl get secret --namespace {{ .Release.Namespace }} {{ template "common.names.fullname" . }} -o jsonpath="{.data.mediawiki-password}" | base64 -d)

   microk8s helm upgrade {{ .Release.Name }} ./mediawiki/ \
     --set mariadb.auth.rootPassword="$MARIADB_ROOT_PASSWORD" \
     --set mariadb.auth.password="$MARIADB_PASSWORD" \
     --set mediawikiPassword="$MEDIAWIKI_PASSWORD"
